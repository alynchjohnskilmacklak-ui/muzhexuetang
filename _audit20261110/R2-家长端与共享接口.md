### 3.1 [P1] 反馈已读口径遗留分叉：1529 条历史差异 + AI 第三口径

`src/lib/classroom-feedback/read-state.ts:8-19`
```ts
export function resolveParentFeedbackReadAt(
  _legacyReadAt: Date | null | undefined,
  receipts: ParentFeedbackReadReceipt[],
) {
  if (receipts.length === 0) return null            // ← 无回执即判未读，legacy 不兜底
  if (receipts.some((receipt) => !receipt.read)) return null
  return receipts.reduce<Date | null>(...)
}
```
